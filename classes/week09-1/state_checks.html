  2. python3 -c 'from models import BankAccount; a = BankAccount("Minsu", 1000); a._balance = -100; print(a.get_balance())'

    The underscore (_balance) is only a convention meaning internal use; it does not block access. Assigning to it directly from outside puts the invalid state -100 straight in. That is why the team rule to use only the public methods (deposit, withdraw, get_balance) matters.밑줄(_balance)은 내부용이라는 약속일 뿐 접근을 막지 않는다. 바깥에서 직접 대입하면 -100이라는 잘못된 상태가 그대로 들어간다. 그래서 공개 메서드(deposit, withdraw, get_balance)만 쓰자는 팀 규칙이 중요하다.

"""Runnable lecture example / 강의용 실행 예제."""
from models import BankAccount
 
a = BankAccount("Minsu", 1000)
for action, amount in [("deposit", 200), ("withdraw", 500),
                       ("withdraw", 900), ("withdraw", -50),
                       ("withdraw", 700)]:
    result = getattr(a, action)(amount)
    print(action, amount, result, a.get_balance())
 
# Check both return and state. / 반환값과 상태를 함께 검사.
a = BankAccount("Minsu", 1000)
assert a.withdraw(1200) is False
assert a.get_balance() == 1000
assert a.withdraw(1000) is True
assert a.get_balance() == 0
print("State checks passed")

"""Week 9 OOP supplement. Run with Python 3.10+; no packages required."""
 
 
class BankAccount:
    def __init__(self, owner, balance=0):
        if not isinstance(owner, str) or not owner.strip():
            raise ValueError("owner must be nonblank text")
        if type(balance) is not int or balance < 0:
            raise ValueError("balance must be a nonnegative integer")
        self.owner = owner.strip()
        self._balance = balance
 
    def get_balance(self):
        return self._balance
 
    def deposit(self, amount):
        if type(amount) is not int or amount <= 0:
            return False
        self._balance += amount
        return True
 
    def withdraw(self, amount):
        if type(amount) is not int or amount <= 0 or amount > self._balance:
            return False
        self._balance -= amount
        return True
 
    def account_type(self):
        return "Standard"
 
 
class SavingsAccount(BankAccount):
    def __init__(self, owner, balance=0, bonus=100):
        super().__init__(owner, balance)
        if type(bonus) is not int or bonus <= 0:
            raise ValueError("bonus must be a positive integer")
        self.bonus = bonus
 
    def add_bonus(self):
        return self.deposit(self.bonus)
 
    def account_type(self):
        return "Savings"
 
 
class AccountBook:
    """Holds references to account objects; never copies their balances."""
    def __init__(self):
        self._accounts = []
 
    def add_account(self, account):
        # Contract: account supports get_balance() and account_type().
        # In this lesson it is a BankAccount or SavingsAccount.
        if account in self._accounts:
            return False
        self._accounts.append(account)
        return True
 
    def total_balance(self):
        total = 0
        for account in self._accounts:
            total += account.get_balance()
        return total
 
    def report(self):
        rows = []
        for account in self._accounts:
            rows.append((account.owner, account.account_type(), account.get_balance()))
        return rows
