"""Study logger: log one study session, then show the weekly totals."""
from logic import make_record, weekly_totals
from report import format_totals
from storage import append_record, load_records
 
DATA = "data/study.csv"
 
 
def main():
    date = input("Date (YYYY-MM-DD): ")
    course = input("Course: ")
    minutes = input("Minutes: ")
    try:
        record = make_record(date, course, minutes)
    except ValueError as error:
        print("Not saved:", error)
    else:
        append_record(DATA, record)
        print("Saved.")
    totals = weekly_totals(load_records(DATA))
    for line in format_totals(totals):
        print(line)
 
 
if __name__ == "__main__":
    main()

"""Rules of the study logger. No input(), print() or open() in this file."""
 
 
def parse_minutes(text):
    """Return text as a positive whole number of minutes.
 
    Raises ValueError if text is not a whole number or is not above 0.
    """
    minutes = int(text)
    if minutes <= 0:
        raise ValueError(f"minutes must be positive, got {minutes}")
    return minutes
 
 
def make_record(date, course, minutes_text):
    """Return one record: {"date": str, "course": str, "minutes": int}.
 
    The course is stored in lower case. Raises ValueError for an empty
    course or invalid minutes.
    """
    course = course.strip().lower()
    if not course:
        raise ValueError("course must not be empty")
    return {"date": date.strip(), "course": course,
            "minutes": parse_minutes(minutes_text)}
 
 
def weekly_totals(records):
    """Return {course: total minutes} for a list of records.
 
    An empty list returns {}. The input list is not changed.
    """
    totals = {}
    for record in records:
        course = record["course"]
        totals[course] = totals.get(course, 0) + record["minutes"]
    return totals

"""Reading and writing study.csv. The only file that touches the disk."""
import csv
from pathlib import Path
 
FIELDS = ["date", "course", "minutes"]
 
 
def load_records(path):
    """Return every row of the CSV file as a record dict.
 
    minutes is converted to int. A missing file returns [].
    """
    path = Path(path)
    if not path.exists():
        return []
    with open(path, newline="", encoding="utf-8") as f:
        return [{"date": row["date"], "course": row["course"],
                 "minutes": int(row["minutes"])}
                for row in csv.DictReader(f)]
 
 
def append_record(path, record):
    """Append one record to the CSV file, writing the header if the file is new."""
    path = Path(path)
    is_new = not path.exists()
    with open(path, "a", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=FIELDS)
        if is_new:
            writer.writeheader()
        writer.writerow(record)

"""Turning results into text for the screen."""
 
 
def format_totals(totals):
    """Return one line per course, largest total first: "python   90 min"."""
    ordered = sorted(totals.items(), key=lambda item: item[1], reverse=True)
    return [f"{course:<8}{minutes:>4} min" for course, minutes in ordered]
